Autonomous Navigation System (SLAM + RTOS)

// Sensor fusion tick — RTOS task
void fusion_task(void* arg) {
  FusionState state;
  while (1) {
    LidarScan scan = lidar_read();
    ImuSample imu   = imu_read();

    state = ekf_update(
      state, scan, imu
    );

    xQueueSend(
      planner_q, &state, 0
    );
    vTaskDelayUntil(
      &last_wake, pdMS_TO_TICKS(10)
    );
  }
}
